/**
 * Protection for the one piece of state that cannot be rebuilt.
 *
 * Everything else in this app lives in Mongo and can be recreated. The WhatsApp
 * credentials cannot: losing them costs a physical QR scan on someone's phone. An unlink
 * was once triggered to test a feature and threw away a working session for no benefit,
 * so the archive-and-restore behaviour is pinned here rather than trusted.
 *
 * C3B (wa-key-store): the same cases now run over every backend through the seam `each()`
 * below: the file store, the database store on an in-memory fake collection, and the
 * database store on a throwaway database of a LOOPBACK mongod (only when C3B_TEST_MONGO_URI
 * is set; otherwise those cases are skipped and printed as NOT-MEASURED, never PASS).

/** C3B-I6: every Mongo-using test passes this first; anything but a loopback host aborts. */
function assertLoopbackMongo(uri: string): void {
  let host = '';
  try { host = new URL(uri).hostname; } catch { /* unparseable: refused below */ }
  if (host !== '127.0.0.1' && host !== 'localhost') {
    throw new Error('C3B-I6: refusing a MONGO_URI whose host is not 127.0.0.1 or localhost');
  }
}

function mulberry32(a: number): () => number {
  return () => {
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

const leaks = (text: string): string[] => {
  const hits: string[] = [];
  SECRETS.forEach((s, i) => {
    for (const form of [s.toString('latin1'), s.toString('hex'), s.toString('base64'), s.toString('base64url')]) {
      if (form.length >= 8 && text.includes(form)) hits.push(`secret#${i}`);
    }
  });
  return hits;
};

async function rejectsWith(p: Promise<unknown>, code: string, label: string) {
  try {
    await p;
  } catch (e) {
    assert.ok(e instanceof KeyStoreError, `${label}: error is a KeyStoreError`);
    assert.equal((e as InstanceType<typeof KeyStoreError>).code, code, `${label}: code`);
    thrown.push(String((e as Error).message));
    return;
  }
  assert.fail(`${label}: expected a rejection with ${code}`);
}

const ALL_TYPES = [
  'pre-key', 'session', 'sender-key', 'sender-key-memory', 'app-state-sync-key',
  'app-state-sync-version', 'lid-mapping', 'device-list', 'tctoken', 'identity-key',
] as const;
type KType = (typeof ALL_TYPES)[number];

function genValue(type: KType, r: () => number): Any {
  const int = (n: number) => Math.floor(r() * n);
  const rb = (n: number) => Buffer.from(Array.from({ length: n }, () => int(256)));
  switch (type) {
    case 'pre-key': return { private: rb(32), public: rb(32) };
    case 'session': return rb(48);
    case 'sender-key': return rb(40);
    case 'sender-key-memory': return { [`${int(1e6)}@s.whatsapp.net`]: true, [`${int(1e6)}@lid`]: true };
    case 'app-state-sync-key':
      return { keyData: rb(32), fingerprint: { rawId: int(1e6), currentIndex: int(100), deviceIndexes: [int(5), int(5)] } };
    case 'app-state-sync-version':
      return { version: int(1000), hash: rb(128), indexValueMap: { [rb(6).toString('hex')]: { valueMac: rb(32) } } };
    case 'lid-mapping': return `${int(1e9)}@lid`;
    case 'device-list': return [String(int(9)), String(int(9))];
    case 'tctoken': return { token: rb(20), timestamp: String(1_700_000_000 + int(1e6)) };
    case 'identity-key': return rb(32);
  }
}

/* ------------------------------------------------------------------ the fake collection and the probe */

function veq(a: Any, b: Any): boolean {
  if (a === undefined) a = null;
  if (b === undefined) b = null;
  if (a === null || b === null) return a === b;
  if (a instanceof ObjectId && b instanceof ObjectId) return a.equals(b);
  if (a instanceof Date && b instanceof Date) return a.getTime() === b.getTime();
  return a === b;
}
const num = (v: Any) => (v instanceof Date ? v.getTime() : v);
const isOpObj = (c: Any) =>
  c !== null && typeof c === 'object' && !(c instanceof ObjectId) && !(c instanceof Date) && !Array.isArray(c) &&
  Object.keys(c).some((k) => k.startsWith('$'));

function matches(doc: Any, f: Any): boolean {
  return Object.entries(f).every(([k, c]: [string, Any]) => {
    const dv = doc[k];
    if (isOpObj(c)) {
      return Object.entries(c).every(([op, arg]: [string, Any]) => {
        switch (op) {
          case '$in': return (arg as Any[]).some((x) => veq(dv, x));
          case '$ne': return !veq(dv, arg);
          case '$gt': return dv != null && num(dv) > num(arg);
          case '$lte': return dv != null && num(dv) <= num(arg);
          default: throw new Error(`fake collection: unsupported operator ${op}`);
        }
      });
    }
    return veq(dv, c);
  });
}

const dupKey = () => Object.assign(new Error('E11000 duplicate key error (fake)'), { code: 11000 });

/** In-memory collection with the unique indexes the real one has (a missing field counts as null, as in Mongo). */
class FakeColl implements KeyColl<Any> {
  docs: Any[] = [];
  constructor(private unique: string[][], private rand: () => number) {}
  private async tick() {
    if (this.rand() < 0.3) await new Promise((r) => setImmediate(r)); else await Promise.resolve();
  }
  private clash(cand: Any, self: Any): boolean {
    return this.unique.some((fields) => this.docs.some((d) => d !== self && fields.every((f) => veq(d[f], cand[f]))));
  }
  private apply(doc: Any, u: Any) {
    for (const [k, v] of Object.entries(u.$set ?? {})) doc[k] = v;
    for (const k of Object.keys(u.$unset ?? {})) delete doc[k];
  }
  async findOne(f: Any) { await this.tick(); const d = this.docs.find((x) => matches(x, f)); return d ? { ...d } : null; }
  find(f: Any) {
    return { toArray: async () => { await this.tick(); return this.docs.filter((x) => matches(x, f)).map((d) => ({ ...d })); } };
  }
  async updateOne(f: Any, u: Any, o?: { upsert?: boolean }) {
    await this.tick();
    const d = this.docs.find((x) => matches(x, f));
    if (d) {
      const next = { ...d }; this.apply(next, u);
      if (this.clash(next, d)) throw dupKey();
      Object.assign(d, next); for (const k of Object.keys(u.$unset ?? {})) delete d[k];
      return { matchedCount: 1 };
    }
    if (!o?.upsert) return { matchedCount: 0 };
    const nd: Any = { _id: new ObjectId() };
    for (const [k, c] of Object.entries(f)) if (!isOpObj(c)) nd[k] = c;
    for (const [k, v] of Object.entries(u.$setOnInsert ?? {})) nd[k] = v;
    this.apply(nd, u);
    if (this.clash(nd, null)) throw dupKey();
    this.docs.push(nd);
    return { matchedCount: 0 };
  }
  async updateMany(f: Any, u: Any) {
    await this.tick();
    for (const d of this.docs.filter((x) => matches(x, f))) {
      const next = { ...d }; this.apply(next, u);
      if (this.clash(next, d)) throw dupKey();
      Object.assign(d, next); for (const k of Object.keys(u.$unset ?? {})) delete d[k];
    }
  }
  async deleteOne(f: Any) { await this.tick(); const i = this.docs.findIndex((x) => matches(x, f)); if (i >= 0) this.docs.splice(i, 1); }
  async deleteMany(f: Any) { await this.tick(); this.docs = this.docs.filter((x) => !matches(x, f)); }
}

/** Wraps any KeyColl (fake or the real collection): records every query, can inject failures. */
class Probe implements KeyColl<Any> {
  calls: { op: string; filter: Any; update?: Any }[] = [];
  failOn: ((op: string, filter: Any, update?: Any) => Error | null) | null = null;
  constructor(public inner: KeyColl<Any>) {}
  private rec(op: string, filter: Any, update?: Any) {
    this.calls.push({ op, filter, update });
    const e = this.failOn?.(op, filter, update);
    if (e) throw e;
  }
  async findOne(f: Any, o?: Any) { this.rec('findOne', f); return this.inner.findOne(f, o); }
  find(f: Any, o?: Any) { this.rec('find', f); return this.inner.find(f, o); }
  async updateOne(f: Any, u: Any, o?: Any) { this.rec('updateOne', f, u); return this.inner.updateOne(f, u, o); }
  async updateMany(f: Any, u: Any) { this.rec('updateMany', f, u); return this.inner.updateMany(f, u); }
  async deleteOne(f: Any) { this.rec('deleteOne', f); return this.inner.deleteOne(f); }
  async deleteMany(f: Any) { this.rec('deleteMany', f); return this.inner.deleteMany(f); }
  /** Unrecorded raw read (the way the contract reads "the raw backend rows, not through the store"). */
  raw(): Promise<Any[]> { return this.inner.find({}).toArray(); }
}

/* ------------------------------------------------------------------ the backends and the test seam */

interface Rig {
  name: 'file' | 'fake' | 'mongo';
  skip: string | undefined;
  creds: Probe | null;
  keys: Probe | null;
  activate(): Promise<void>;
  deactivate(): Promise<void>;
  wipe(): Promise<void>;
}

const realNow = () => new Date();
const NOT_MEASURED = 'NOT-MEASURED: no loopback mongod (C3B_TEST_MONGO_URI unset)';

function dbRig(name: 'fake' | 'mongo'): Rig {
  const rig: Rig = {
    name,
    skip: name === 'mongo' && !MONGO_URI ? NOT_MEASURED : undefined,
    creds: null,
    keys: null,
    async activate() {
      cfg.keyStore = 'mongo';
      cfg.keyEncryptionKey = KEY1;
      seam.now = realNow;
      if (name === 'fake') {
        const rnd = mulberry32(SEED ^ 0x5eed);
        rig.creds = new Probe(new FakeColl([['ownerUserId', 'archiveId']], rnd));
        rig.keys = new Probe(new FakeColl([['ownerUserId', 'archiveId', 'type', 'id']], rnd));
      } else {
        rig.creds = new Probe(mongo.waCreds() as unknown as KeyColl<Any>);
        rig.keys = new Probe(mongo.waKeys() as unknown as KeyColl<Any>);
      }
      seam.creds = rig.creds;
      seam.keys = rig.keys;
    },
    async deactivate() {
      cfg.keyStore = 'file';
      cfg.keyEncryptionKey = null;
      seam.creds = null;
      seam.keys = null;
      seam.now = realNow;
    },
    async wipe() {
      await rig.creds!.inner.deleteMany({});
      await rig.keys!.inner.deleteMany({});
    },
  };
  return rig;
}

const fileRig: Rig = {
  name: 'file',
  skip: undefined,
  creds: null,
  keys: null,
  async activate() { cfg.keyStore = 'file'; cfg.keyEncryptionKey = null; },
  async deactivate() { cfg.keyStore = 'file'; },
  async wipe() { await rm(ROOT, { recursive: true, force: true }); },
};
const fakeRig = dbRig('fake');
const mongoRig = dbRig('mongo');
const RIGS: Rig[] = [fileRig, fakeRig, mongoRig];

let current: Rig = fileRig;
const isDb = () => current.name !== 'file';

/**
 * THE TEST SEAM: one test body, registered once per backend. `each(name, body)` runs the body
 * on the file store, on the database store over the in-memory fake, and on the database store
 * over a loopback mongod (skipped as NOT-MEASURED when there is none).
 */
function each(name: string, body: (rig: Rig) => Promise<void>, only?: Array<Rig['name']>, timeout?: number) {
  for (const rig of RIGS) {
    if (only && !only.includes(rig.name)) continue;
    const opts: { skip?: string; timeout?: number } = {};
    if (rig.skip) opts.skip = rig.skip;
    if (timeout) opts.timeout = timeout;
    test(`${name} [${rig.name}]`, opts, async () => {
      current = rig;
      if (rig.name === 'mongo') assertLoopbackMongo(config.mongoUri);
      await rig.activate();
      try { await body(rig); } finally { await rig.deactivate(); current = fileRig; }
    });
  }
}

/* ------------------------------------------------------------------ backend-neutral helpers for the existing cases */

async function fakeLink(userId: string, marker = 'original') {
  if (isDb()) {
    const { state, saveCreds } = await createAuthState(userId);
    (state.creds as Any).marker = marker;
    await saveCreds();
    await state.keys.set({ 'app-state-sync-key': { AAA: {} as Any } });
    return;
  }
  const dir = sessionDir(userId);
  await mkdir(dir, { recursive: true });
  await writeFile(join(dir, 'creds.json'), JSON.stringify({ marker }));
  await writeFile(join(dir, 'app-state-sync-key-AAA.json'), '{}');
}

const exists = async (p: string) => {
  if (isDb()) {
    // the database has no paths: ".../<owner>/creds.json" is the live creds row, "<archive>/creds.json" an archived one
    const dir = basename(dirname(p));
    const owner = /^[0-9a-f]{24}$/.test(dir) ? dir : USER;
    const archiveId = owner === dir ? null : dir;
    return (await current.creds!.raw()).some((r) => r.ownerUserId.toHexString() === owner && r.archiveId === archiveId);
  }
  try { await access(p); return true; } catch { return false; }
};

async function readCredsText(userId: string): Promise<string> {
  if (isDb()) {
    const { state } = await createAuthState(userId);
    return JSON.stringify(state.creds, BufferJSON.replacer);
  }
  return readFile(join(sessionDir(userId), 'creds.json'), 'utf8');
}

async function halfLink(userId: string) {
  if (isDb()) {
    const { state } = await createAuthState(userId);
    await state.keys.set({ 'pre-key': { '1': { private: randomBytes(32), public: randomBytes(32) } } });
    return;
  }
  await mkdir(sessionDir(userId), { recursive: true });
}

async function archiveNames(userId: string): Promise<string[]> {
  if (isDb()) {
    const rows = [...(await current.creds!.raw()), ...(await current.keys!.raw())];
    return [...new Set(rows.filter((r) => r.ownerUserId.toHexString() === userId && r.archiveId !== null).map((r) => r.archiveId as string))];
  }
  return (await readdir(join(ROOT, '.trash'))).filter((n) => n.startsWith(userId));
}

async function connectedWipe() {
  if (!MONGO_URI) return;
  await mongo.waCreds().deleteMany({});
  await mongo.waKeys().deleteMany({});
}

before(async () => {
  await rm(ROOT, { recursive: true, force: true });
  if (MONGO_URI) {
    assertLoopbackMongo(config.mongoUri);
    await mongo.connect();
    await mongo.ensureIndexes();
  }
});
beforeEach(async () => { await rm(ROOT, { recursive: true, force: true }); await connectedWipe(); });
after(async () => {
  await rm(ROOT, { recursive: true, force: true });
  if (MONGO_URI) {
    const db = await mongo.connect();
    assert.ok(db.databaseName.startsWith('whatsapp_msg_test_'), 'only a throwaway database is dropped');
    await db.dropDatabase();
    await mongo.close();
  }
  // C3B-I4: nothing this module logged, and no error it raised, carries a key, a value or the canary.
  assert.deepEqual(leaks(captured.join('\n')), [], 'a secret reached a log line');
  assert.deepEqual(leaks(thrown.join('\n')), [], 'a secret reached an error message');
});

/* ================================================================== the existing cases, over every backend */

each('unlinking archives the credentials instead of deleting them', async () => {
  await fakeLink(USER);
  const archived = await destroyAuthState(USER);

  assert.ok(archived, 'an archive path must be returned');
  assert.equal(await exists(join(sessionDir(USER), 'creds.json')), false, 'live link is gone');
  assert.equal(await exists(join(archived!, 'creds.json')), true, 'but the credentials survive');
  assert.equal(await hasRestorableAuthState(USER), true);
});

each('an accidental unlink can be undone', async () => {
  await fakeLink(USER, 'the-real-session');
  await destroyAuthState(USER);

  assert.equal(await restoreAuthState(USER), true);
  const restored = await readCredsText(USER);
  assert.match(restored, /the-real-session/, 'the original credentials must come back');
});

each('restore refuses to clobber a link that is already live', async () => {
  await fakeLink(USER, 'old');
  await destroyAuthState(USER);
  await fakeLink(USER, 'new-and-working');

  assert.equal(await restoreAuthState(USER), false, 'must not overwrite a working session');
  const live = await readCredsText(USER);
  assert.match(live, /new-and-working/);
});

each('restore reports failure when there is nothing archived', async () => {
  assert.equal(await hasRestorableAuthState(USER), false);
  assert.equal(await restoreAuthState(USER), false);
});

each('a half-finished link with no credentials is simply removed, not archived', async () => {
  await halfLink(USER);
  const archived = await destroyAuthState(USER);
  assert.equal(archived, null);
  assert.equal(await hasRestorableAuthState(USER), false);
});

each('archived links are never mistaken for live sessions on boot', async () => {
  await fakeLink(USER);
  await destroyAuthState(USER);
  // The trash directory sits inside sessions/, so rehydration must skip it or the server
  // would try to bring an unlinked account back online.
  assert.deepEqual(await linkedUserIds(), []);

  await fakeLink(USER);
  assert.deepEqual(await linkedUserIds(), [USER]);
});

each('old archives are pruned so the trash cannot grow without bound', async () => {
  for (let i = 0; i < 8; i += 1) {
    await fakeLink(USER, `gen-${i}`);
    await destroyAuthState(USER);
    await new Promise((r) => setTimeout(r, 5)); // distinct timestamps
  }
  const kept = await archiveNames(USER);
  assert.ok(kept.length <= 5, `expected at most 5 archives, found ${kept.length}`);
});

each('C3B-C1a: 1000 seeded random operations give the same result on the file store, the database store and a Map model', async (rig) => {
  const seed = SEED + 101;
  console.log(`C3B-C1a [${rig.name}] seed=${seed}`);
  const r = mulberry32(seed);
  const int = (n: number) => Math.floor(r() * n);
  const IDS = ['1', '2', '3', '44', 'abc', 'x:y/z', '7', '8', '9', '10', '11', '12'];

  // the reference: the real file store, in its own folder, under the file backend
  cfg.keyStore = 'file';
  const ref = await createAuthState(OWNER_A);
  cfg.keyStore = rig.name === 'file' ? 'file' : 'mongo';
  const sut = rig.name === 'file' ? await createAuthState(OWNER_B) : await createAuthState(OWNER_A);
  const model = new Map<string, Any>();
  const mk = (t: string, id: string) => `${t}\u0000${id}`;

  for (let i = 0; i < 1000; i += 1) {
    const type = ALL_TYPES[int(ALL_TYPES.length)]!;
    if (r() < 0.5) {
      const data: Any = {};
      for (let j = 0, n = 1 + int(3); j < n; j += 1) {
        const t = ALL_TYPES[int(ALL_TYPES.length)]!;
        const id = IDS[int(IDS.length)]!;
        const roll = r();
        const v = roll < 0.25 ? null : t === 'lid-mapping' && roll < 0.3 ? '' : genValue(t, r);
        (data[t] ??= {})[id] = v;
      }
      await ref.state.keys.set(data);
      await sut.state.keys.set(data);
      for (const t of Object.keys(data)) for (const id of Object.keys(data[t])) {
        const v = data[t][id];
        if (v) model.set(mk(t, id), rt(v)); else model.delete(mk(t, id)); // the file store drops every falsy value
      }
    } else {
      const ids = [...new Set(Array.from({ length: 1 + int(4) }, () => IDS[int(IDS.length)]!))];
      const a = await ref.state.keys.get(type, ids);
      const b = await sut.state.keys.get(type, ids);

/** Every leaf of a raw row as a Buffer, plus the JSON text, so a plaintext hit anywhere is found. */
function leafBlobs(row: Any): Buffer[] {
  const out: Buffer[] = [];
  const walk = (v: Any) => {
    if (v instanceof Binary) out.push(bin(v));
    else if (Buffer.isBuffer(v)) out.push(v);
    else if (typeof v === 'string') out.push(Buffer.from(v));
    else if (v instanceof ObjectId) out.push(Buffer.from(v.toHexString()));
    else if (v instanceof Date) out.push(Buffer.from(v.toISOString()));
    else if (v && typeof v === 'object') Object.values(v).forEach(walk);
  };
  walk(row);
  return out;
}
function canaryNeedles(c: string): Buffer[] {
  const b64 = (pad: number) => Buffer.from('x'.repeat(pad) + c).toString('base64');
  return [Buffer.from(c), Buffer.from(Buffer.from(c).toString('hex')), Buffer.from(b64(0).slice(0, -4)), Buffer.from(b64(1).slice(2, -4)), Buffer.from(b64(2).slice(3, -4))];
}
const hasCanary = (rows: Any[]) => rows.some((row) => leafBlobs(row).some((b) => canaryNeedles(CANARY).some((n) => b.includes(n))));

const allCarryOwner = (rig: Rig, from: { c: number; k: number }) => {
  const bad: string[] = [];
  rig.creds!.calls.slice(from.c).forEach((c) => { if (c.filter.ownerUserId === undefined) bad.push(`creds.${c.op} ${JSON.stringify(c.filter)}`); });
  rig.keys!.calls.slice(from.k).forEach((c) => { if (c.filter.ownerUserId === undefined) bad.push(`keys.${c.op} ${JSON.stringify(c.filter)}`); });
  return bad;
};
