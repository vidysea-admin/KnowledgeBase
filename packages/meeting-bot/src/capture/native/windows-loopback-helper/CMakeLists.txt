cmake_minimum_required(VERSION 3.24)
project(lkb_windows_loopback LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Windows process-loopback only")
endif()
add_executable(lkb-windows-loopback main.cpp)
target_compile_features(lkb-windows-loopback PRIVATE cxx_std_20)
target_compile_definitions(lkb-windows-loopback PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(lkb-windows-loopback PRIVATE ole32 mmdevapi ksuser uuid)
if(MSVC)
  target_compile_options(lkb-windows-loopback PRIVATE /W4 /WX /guard:cf /sdl)
  target_link_options(lkb-windows-loopback PRIVATE /DYNAMICBASE /NXCOMPAT /guard:cf)
endif()
